import { NextResponse } from "next/server";
import { getStore } from "@netlify/blobs";

const PUSH_STORE_NAME = "push-subscriptions";

export async function POST(request: Request) {
  try {
    const subscription = await request.json();

    if (!subscription?.endpoint) {
      return NextResponse.json(
        { error: "Invalid push subscription" },
        { status: 400 },
      );
    }

    const store = getStore(PUSH_STORE_NAME);

    // 使用 endpoint 生成稳定的 key，避免同一设备重复保存。
    const key = Buffer.from(subscription.endpoint).toString("base64url");

    await store.setJSON(key, subscription);

    console.log("[Push] subscription saved:", subscription.endpoint);

    return NextResponse.json({ ok: true });
  } catch (error) {
    console.error("[Push] Failed to save subscription:", error);

    return NextResponse.json(
      { error: "Failed to save push subscription" },
      { status: 500 },
    );
  }
}
